# AIGC 平台核心模块设计文档：密钥池与任务执行系统 (v3.0)

**核心设计哲学**：

1. **解耦**：密钥管理模块不关心业务，只管分发凭证。
2. **异步**：API 负责生产，Worker 负责消费，中间通过 Redis 缓冲。
3. **高可用**：自动熔断坏 Key，自动负载均衡。

## 二、 Redis 数据结构设计

| 模块 | Key 格式 | 类型 | TTL | 作用 |
| --- | --- | --- | --- | --- |
| **密钥状态** | `pool:usage:{key_id}` | String (Int) | 24h | 当前并发计数 |
| **密钥熔断** | `pool:cooldown:{key_id}` | String | 300s | 熔断标记 (存在即不可用) |
| **密钥统计** | `pool:stats:calls:{key_id}` | String (Int) | - | 调用次数缓冲 (Write-Behind) |
| **任务进度** | `task:progress:{task_id}` | String (Int) | 24h | **实时进度条** (Worker写, API读) |
| **等待队列** | `queue:waiting:vip` | List | - | **VIP 任务候车室** (T4-T5) |
| **等待队列** | `queue:waiting:normal` | List | - | **普通任务候车室** (T1-T3) |
| **执行队列** | `queue:runnable` | List | - | **发车区** (已分配Key，Worker监听) |

## 三、 核心模块实现：KeyManager

该类负责所有关于 Key 的调度、分发和回收。

```python
# app/services/key_manager.py
import random
import json
import logging
from app.extensions import db, redis_client
from app.models.key import ApiKey
from app.models.task import Task
from sqlalchemy import text

logger = logging.getLogger(__name__)

class KeyManager:
    # ... 前缀常量定义 ...
    
    @classmethod
    def allocate_key(cls, platform: str):
        """
        [内部方法] 尝试获取一个可用 Key
        逻辑: 查库 -> 过滤冷却 -> 权重随机 -> 原子占用
        """
        # 1. 查出该平台所有启用的 Key
        candidates = ApiKey.query.filter_by(platform=platform, status=1).all()
        valid = [k for k in candidates if not redis_client.get(f"pool:cooldown:{k.id}")]
        
        if not valid: return None
        
        # 2. 权重随机尝试
        weights = [k.weight for k in valid]
        # 尝试次数 = 候选数 (避免死循环)
        for _ in range(len(valid)):
            selected = random.choices(valid, weights=weights, k=1)[0]
            
            # 3. 原子占用
            curr = redis_client.incr(f"pool:usage:{selected.id}")
            if curr <= selected.max_concurrency:
                redis_client.incr(f"pool:stats:calls:{selected.id}") # 统计+1
                return selected
            else:
                redis_client.decr(f"pool:usage:{selected.id}") # 回滚
                continue
                
        return None

    @classmethod
    def release_key_and_dispatch(cls, key_id: int):
        """
        [核心] 释放 Key 占用，并触发事件分发
        """
        # 1. 释放占用
        r_key = f"pool:usage:{key_id}"
        curr = redis_client.get(r_key)
        if curr and int(curr) > 0:
            redis_client.decr(r_key)
            
        # 2. 【事件驱动】主动去等待队列捞人
        cls._dispatch_next_task(key_id)

    @classmethod
    def _dispatch_next_task(cls, key_id: int):
        """
        [调度器] 优先从 VIP 队列捞取任务
        """
        # 1. 检查 Key 是否还健康/有空闲 (Double Check)
        key_obj = ApiKey.query.get(key_id)
        if not key_obj or redis_client.get(f"pool:cooldown:{key_id}"): return
        
        usage = redis_client.get(f"pool:usage:{key_id}")
        if usage and int(usage) >= key_obj.max_concurrency: return

        # 2. 【VIP 插队逻辑】先看 VIP，再看普通
        queue_vip = "queue:waiting:vip"
        queue_norm = "queue:waiting:normal"
        
        raw_task = redis_client.lpop(queue_vip)
        if not raw_task:
            raw_task = redis_client.lpop(queue_norm)
            
        if not raw_task: return # 没人排队，休息

        # 3. 【惰性取消检查】
        task_data = json.loads(raw_task)
        task_id = task_data['task_id']
        
        # 查库看状态：如果用户取消了，直接丢弃，递归找下一个
        task_status = db.session.query(Task.status).filter_by(id=task_id).scalar()
        if task_status == 'cancelled':
            logger.info(f"Task {task_id} cancelled. Skipping.")
            cls._dispatch_next_task(key_id) # 递归
            return

        # 4. 注入 Key 信息并推送到执行队列
        # 再次占用
        redis_client.incr(f"pool:usage:{key_id}")
        redis_client.incr(f"pool:stats:calls:{key_id}")
        
        task_data['key_id'] = key_obj.id
        task_data['api_base'] = key_obj.api_base # 注入 Base
        task_data['api_key'] = key_obj.api_key   # 注入 Key
        
        redis_client.rpush("queue:runnable", json.dumps(task_data))
        
        # 5. 如果还有并发余额，继续捞 (填满为止)
        cls._dispatch_next_task(key_id)

    @classmethod
    def mark_unhealthy(cls, key_id: int, reason: str):
        """触发熔断"""
        redis_client.setex(f"pool:cooldown:{key_id}", 300, "1")
        # 注意：这里只做标记，release 由 worker 的 finally 块统一调用

```

## 四、 业务流程集成

### 1. 任务提交 (API Layer)

文件: `app/services/task_service.py`

```python
def submit_task(user, task_data):
    # 1. 构造基础 Payload
    payload = {
        "task_id": generate_uuid(),
        "params": task_data['params'],
        "platform": task_data['model_platform']
    }

    # 2. 尝试直接获取资源 (快车道)
    key = KeyManager.allocate_key(task_data['model_platform'])
    
    if key:
        # 有资源 -> 注入 Key -> 进执行队列
        payload['key_id'] = key.id
        payload['api_base'] = key.api_base
        payload['api_key'] = key.api_key
        redis_client.rpush("queue:runnable", json.dumps(payload))
        msg = "任务开始执行"
    else:
        # 无资源 -> 进等待队列 (慢车道)
        # 根据等级分流
        target_queue = "queue:waiting:vip" if user.level >= 4 else "queue:waiting:normal"
        redis_client.rpush(target_queue, json.dumps(payload))
        msg = "当前排队人数较多，已进入优先等待序列" if user.level >= 4 else "正在排队中"

    return {"status": "pending", "msg": msg}

```

### 3. 任务执行 (Worker Layer)

文件: `worker.py`
Worker 从 `queue:runnable` 取出的 JSON **一定**包含 `api_base` 和 `api_key`。

```python
import time
import requests
from app.extensions import db, redis_client
from app.services.key_manager import KeyManager

def process_task(payload):
    task_id = payload['task_id']
    key_id = payload['key_id']
    api_base = payload['api_base'] # 从 Payload 拿
    api_key = payload['api_key']   # 从 Payload 拿
    
    # 1. 更新 DB 为 Processing
    task = Task.query.get(task_id)
    task.status = 'processing'
    db.session.commit()

    try:
        # 2. 调用上游 (使用 Payload 里的 api_base)
        headers = {"Authorization": f"Bearer {api_key}"}
        
        # Submit ...
        submit_url = f"{api_base.rstrip('/')}/videos/generations"
        resp = requests.post(submit_url, headers=headers, json=payload['params'])
        
        # 错误处理
        if resp.status_code in [401, 429]:
            KeyManager.mark_unhealthy(key_id, f"HTTP {resp.status_code}")
            raise Exception("Provider Error")
            
        task_uuid = resp.json()['id']

        # 3. 轮询进度
        while True:
            time.sleep(3)
            # Check Status ...
            status_url = f"{api_base.rstrip('/')}/videos/generations/{task_uuid}"
            check = requests.get(status_url, headers=headers)
            
            data = check.json()
            state = data.get('status') # SUCCEEDED / FAILED / RUNNING
            progress = data.get('progress', 0)
            
            # 【核心】写 Redis 进度 (供前端轮询)
            if state == 'RUNNING':
                redis_client.setex(f"task:progress:{task_id}", 86400, progress)
                
            elif state == 'SUCCEEDED':
                task.status = 'success'
                task.result_url = data['output']['url']
                task.finished_at = datetime.now()
                # 清理 Redis 进度
                redis_client.delete(f"task:progress:{task_id}")
                db.session.commit()
                break
                
            elif state == 'FAILED':
                raise Exception(data.get('failure_reason'))

    except Exception as e:
        task.status = 'failed'
        task.fail_reason = str(e)
        redis_client.delete(f"task:progress:{task_id}")
        db.session.commit()
        
    finally:
        # 【核心】释放 Key 并触发下一个任务
        KeyManager.release_key_and_dispatch(key_id)

```

## 五、 定时维护任务

为了系统的稳健性，需要挂载两个后台任务：

1. **统计同步 (Sync Stats)**: 每 5 分钟，将 Redis 里的 `pool:stats:calls` 累加到 MySQL，并清零 Redis。
2. **看门狗 (Watchdog)**: 每 1 分钟，检查 `queue:waiting:*` 是否有积压且有 Key 空闲（防止 `release` 事件丢失导致的队列死锁）。

## 六、 方案总结

这套方案完美响应了您对**“多渠道中转”**和**“精细化控制”**的需求：

1. **多渠道支持**: `api_base` 存储在 Key 配置中，Worker 动态读取，您可以同时混用 OpenAI 官方 Key 和 第三方中转 Key。
2. **VIP 尊享**: 高等级用户进 `waiting:vip`，Key 释放时优先服务，体验极佳。
3. **无损取消**: 用户取消仅改 DB，系统自动过滤，不浪费算力。
4. **实时反馈**: Redis 承载进度条高频读写，数据库只存结果，性能最优。
5. **永不暂停**: API 提交即返回，Worker 永不空转，全流程事件驱动。
